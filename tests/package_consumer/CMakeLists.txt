cmake_minimum_required(VERSION 3.21)
project(NovaInfraInstalledConsumer LANGUAGES CXX)
find_package(NovaInfra CONFIG REQUIRED)
add_executable(consumer main.cpp)
set_target_properties(consumer PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
target_compile_options(consumer PRIVATE -pedantic-errors)
foreach(module IN ITEMS Core Logger Config ThreadPool SqlitePool EventBus StateMachine)
    if(TARGET NovaInfra::${module})
        target_link_libraries(consumer PRIVATE NovaInfra::${module})
        string(TOUPPER "${module}" module_upper)
        string(REPLACE "_" "" module_normalized "${module_upper}")
        target_compile_definitions(consumer PRIVATE CONSUMER_${module_normalized})
    endif()
endforeach()
